cmake_minimum_required(VERSION 3.16)
project(bag_cloud_viewer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets Gui)
find_package(OpenGL REQUIRED)
find_package(BZip2 REQUIRED)
find_library(LZ4_LIBRARY NAMES lz4 REQUIRED)
add_library(bag_reader src/bag.cpp src/accumulation.cpp src/odometry.cpp)
target_include_directories(bag_reader PUBLIC src)
target_link_libraries(bag_reader PRIVATE BZip2::BZip2 ${LZ4_LIBRARY})
target_compile_options(bag_reader PRIVATE -Wall -Wextra -Wpedantic)
add_executable(bag_cloud_viewer src/main.cpp)
target_link_libraries(bag_cloud_viewer PRIVATE bag_reader Qt5::Widgets Qt5::Gui OpenGL::GL)
target_compile_options(bag_cloud_viewer PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(reader_tests tests/reader_tests.cpp)
target_link_libraries(reader_tests PRIVATE bag_reader BZip2::BZip2 ${LZ4_LIBRARY})
add_test(NAME reader_tests COMMAND reader_tests)
